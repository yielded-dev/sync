## The code behind the demo

Define the source once. The server decides each reservation; clients predict it.

<Tabs>
  <TabItem label="seats.ts">

```ts
import { Action, Source } from "@yielded/sync";
import { Schema } from "effect";

export class SoldOut extends Schema.TaggedError<SoldOut>()("SoldOut", {}) {}

export const Seats = Source.make({
  kind: "seats",
  id: Schema.NonEmptyString.pipe(Schema.brand("ShowId")),
  schemaVersion: 1,
  snapshot: Schema.Struct({ seats: Schema.Array(Schema.NullOr(Schema.String)) }),
  event: Schema.Struct({ seat: Schema.Int, guest: Schema.String }),
  actions: [
    Action.make("reserve", {
      payload: Schema.Struct({ guest: Schema.String }),
      success: Schema.Struct({ seat: Schema.Int }),
      error: SoldOut,
    }),
  ],
});
```

  </TabItem>
  <TabItem label="seats-server.ts">

```ts
import { Server } from "@yielded/sync/server";
import { Effect } from "effect";

import { authorize, Principal } from "./auth";
import { Seats, SoldOut } from "./seats";

export const SeatsServer = Server.make(Seats, {
  principal: Principal,
  state: Seats.spec.snapshot,
  initialize: Effect.succeed({ seats: [null, null, null] }),
  snapshot: (state) => state,
  authorize,
  actions: {
    reserve: Effect.fn(function* ({ state, payload }) {
      const seat = state.seats.indexOf(null);

      if (seat === -1) return yield* SoldOut.make({});
      const seats = state.seats.map((guest, index) => (index === seat ? payload.guest : guest));

      return { state: { seats }, events: [{ seat, guest: payload.guest }], result: { seat } };
    }),
  },
});
```

  </TabItem>
  <TabItem label="seats-client.ts">

```ts
import { Client } from "@yielded/sync/client";

import { Seats } from "./seats";

export const SeatsClient = Client.definition(Seats, {
  applyEvent: ({ seats }, event) => ({
    seats: seats.map((guest, index) => (index === event.seat ? event.guest : guest)),
  }),
  optimistic: {
    // Predict the next free seat. The server's event confirms or replaces it.
    reserve: ({ seats }, payload) => {
      const seat = seats.indexOf(null);

      return {
        seats: seats.map((guest, index) => (index === seat ? payload.guest : guest)),
      };
    },
  },
});
```
